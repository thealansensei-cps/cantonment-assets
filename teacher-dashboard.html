🔌 Connect Google Sheets (5 minutes)

1
Go to sheets.google.com and create a new blank spreadsheet. Name it "P4 Magic Spells Progress".
2
In the sheet, click Extensions → Apps Script.
3
Delete all existing code and paste this script:
// P4 Magic Spells – Google Apps Script // Paste this in Extensions > Apps Script function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet() .getSheetByName('Progress') || SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); // Add headers if empty if (sheet.getLastRow() === 0) { sheet.appendRow([ 'Timestamp','Student Name','Class','Game', 'Level Reached','XP Earned','Tasks Done', 'Tasks Total','Self Rating','Reflection' ]); } var data = JSON.parse(e.postData.contents); sheet.appendRow([ data.timestamp, data.studentName, data.classGroup, data.gameName, data.levelReached, data.xpEarned, data.tasksCompleted, data.tasksTotal, data.selfRating, data.reflection ]); return ContentService.createTextOutput('OK') .setMimeType(ContentService.MimeType.TEXT); } function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet() .getSheetByName('Progress') || SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = sheet.getDataRange().getValues(); return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
4
Click Deploy → New deployment. Set type to "Web App". Set "Who has access" to "Anyone". Click Deploy and authorize.
5
Copy the Web App URL and paste it below:

📌 How to embed in Google Sites

1
Host student-tracker.html somewhere (Google Drive → Share → Anyone can view, then get the preview link, or use GitHub Pages).
2
In Google Sites, click Insert → Embed and paste the URL. Resize to fit your page.
3
Open student-tracker.html and replace YOUR_APPS_SCRIPT_URL_HERE with your deployed URL.
4
You can embed the same tracker on each game page — it auto-detects the game name from the URL!
